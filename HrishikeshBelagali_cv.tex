\cvsect{Open Source Contributions}
\begin{entrylist}
	\entry{propaq \\ Python, Rust}
		{propaq}
		{\href{https://github.com/hkbelagali/propaq}{github.com}}
		{Primary developer of \textit{propaq}, a Python package for Heisenberg propagation, designed for large-scale simulation on HPC clusters.
		\begin{itemize}[noitemsep,topsep=0pt,parsep=0pt,partopsep=0pt, leftmargin=7pt]
			\item Implmented thread-safe kernels and compact representation of Pauli and Majorana strings in Rust for optimal multi-threaded simulation.
			\item Developed a Python interface integrating \textit{propaq} with Qiskit and Cirq for seamless simulation workflows using pyo3 bindings.
			\item Designed a plugin system for developing custom noise and truncation policies in Python, Rust, C, and Julia
			to enable user customization with minimal overhead.
		\end{itemize}
		}

		{ffsim}

		{Contributed to \textit{ffsim}, a Python package for simulating fermionic quantum systems.
